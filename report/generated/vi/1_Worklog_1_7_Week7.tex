\vspace{0.5em}\noindent\textbf{Mục tiêu tuần 7:}\par

\begin{itemize}
\tightlist
\item
  Hiểu và áp dụng các chiến lược Tối ưu hóa chi phí (Cost Optimization)
  trên AWS.
\item
  Tìm hiểu quy trình Hiện đại hóa ứng dụng (App Modernization) với
  Serverless và kiến trúc hướng sự kiện (Event-Driven).
\item
  Thực hành xây dựng một ứng dụng hoàn chỉnh (Sách) sử dụng kiến trúc
  Serverless, tích hợp backend, frontend và xác thực người dùng.
\end{itemize}
