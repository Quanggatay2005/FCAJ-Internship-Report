\vspace{0.5em}\noindent\textbf{Nội Dung Nổi Bật}\par

\vspace{0.5em}\noindent\textbf{Chia sẻ về hành trình từ một sinh viên đến AWS Community
Builder (anh
Nghị)}\par

\begin{itemize}
\tightlist
\item
  Bắt đầu từ sinh viên -\textgreater{} tham gia chương trình FCJ
  -\textgreater{} AWS Community Builder
\item
  Được sự hướng dẫn tận tình của các anh chị bên chương trình
\item
  Trải qua nhiều thử thách, mỗi thử thách đều có thể đem lại những bài
  học quý báu
\end{itemize}

\vspace{0.5em}\noindent\textbf{Vai trò, công việc của một DevOps Engineer trong một công ty
(anh
Trọng)}\par

\begin{itemize}
\tightlist
\item
  \textbf{Triển vọng của DevOps trong nền công nghiệp}: múc lương của
  DevOps Engineer thuộc dạng trung bình trong thị trường việc làm IT
  hiện nay, trải dài trên số năm kinh nghiệm
\item
  \textbf{DevOps Engineer làm gì?}: tầm vực của một DevOps Engineer rất
  rộng, tùy thuộc vào quy mô, yêu cầu của từng công ty.
\item
  \textbf{Sinh viên cần làm gì để trở thành một DevOps Engineer?}: anh
  Trọng khuyên mọi người hãy học thật chắc những kiến thức căn bản
  trước.

  \begin{itemize}
  \tightlist
  \item
    Nhập môn lập trình: linux, mạng, ngôn ngữ lập trình, Git, CI/CD,
    containers
  \item
    Hiểu và thực hiện được các ứng dụng đang chạy như thế nào
  \item
    Thực chiến mọi lúc mọi nơi, bất kể khi nào học một thứ gì đó mới.
  \end{itemize}
\end{itemize}

\vspace{0.5em}\noindent\textbf{Chia sẻ của một Data Analytics Engineer về công việc của họ
hàng ngày (anh Dat
Pham)}\par

\begin{itemize}
\tightlist
\item
  \textbf{Công việc hàng ngày của một Data Analytics Engineer}: tùy theo
  domain, mô hình kinh doanh và phòng ban hỗ trợ. Đối với công ty
  Kalmereo, anh Dat sẽ làm:

  \begin{itemize}
  \tightlist
  \item
    Xây dựng báo cáo theo ngày, tuần, tháng, quý để theo dõi
  \item
    Thiết kế Dashboard
  \item
    Phân tích chỉ số kinh doanh
  \item
    Phối hợp các phòng ban để giải quyết vấn đề phát sinh.
  \end{itemize}
\item
  \textbf{Những kĩ năng cần thiết để sống sót}:

  \begin{itemize}
  \tightlist
  \item
    Tư duy phản biện
  \item
    Kỹ năng giao tiếp
  \item
    Kỹ năng kể chuyện với dữ liệu
  \item
    Kỹ năng giải quyết vấn đề
  \end{itemize}
\item
  \textbf{Tư duy phát triển nghề nghiệp}: anh Đạt đề xuất một quy trình
  tuyến tính để có thể từng bước nâng cao năng lực bản thân.

  \begin{itemize}
  \tightlist
  \item
    Người thực thi
  \item
    Người học chủ động
  \item
    Người giải quyết vấn đề
  \item
    Người tư duy hệ thống
  \item
    Người dẫn dắt
  \end{itemize}
\end{itemize}

\vspace{0.5em}\noindent\textbf{Những Gì Học
Được}\par

\begin{itemize}
\tightlist
\item
  \textbf{Đổi mới tư duy}: Tư duy chủ động, sẵn sàng học hỏi, tham gia
  các hoạt động chuyên môn
\item
  \textbf{Kiến thức nền phải thật chắc}: việc chạy đua học các tools
  không giúp tăng khả năng sinh tồn. Ngược lại, những kiến thức cốt lõi,
  nền tảng sẽ giúp bạn thích nghi tốt hơn với sự thay đổi của công nghệ
\item
  \textbf{Thích ứng với sự thay đổi của thời đại mới}: trong thời đại
  mới, mỗi con người cũng cần có thái độ đúng đắn.
\end{itemize}

\vspace{0.5em}\noindent\textbf{Trải nghiệm trong
event}\par

Tham gia workshop là một trải nghiệm rất bổ ích, giúp tôi có cái nhìn
toàn diện về cách mà các vị trí Data Analytics, DevOps Engineer đã làm
trong công việc. Một số trải nghiệm nổi bật:

\vspace{0.5em}\noindent\textbf{Học hỏi từ các diễn giả có chuyên môn
cao}\par

\begin{itemize}
\tightlist
\item
  Được hiểu thêm về những công việc và những bài toán thực tế mà họ đã
  trải qua, giúp tôi có thêm dữ liệu để quyết định hướng đi của mình sau
  này.
\item
  Được học hỏi cách suy nghĩ vấn đề, xử lý vấn đề và đưa ra các giải
  pháp để giải quyết vấn đề. Mỗi diễn giả lại có một cách tiếp cận vấn
  đề rất khác nhau, từ đó giúp tôi có thêm nhiều góc nhìn và lựa chọn
  hơn cho mình.
\end{itemize}
